19792:     function refineImpact(origin, velocity, tLow, tHigh, out) {
19793:       let low = tLow;

19796:         ballisticPoint(origin, velocity, t, tmpBallistic);
19797:         if (tmpBallistic.y > surfaceHeightAt(tmpBallistic.x, tmpBallistic.z)) {
19798:           low = t;
19799:           continue;
19800:         }
19801:         let high = t;
19802:         for (let k = 0; k < 16; k += 1) {
19803:           const mid = (low + high) * 0.5;
19804:           ballisticPoint(origin, velocity, mid, tmpBallistic);
19805:           if (tmpBallistic.y > surfaceHeightAt(tmpBallistic.x, tmpBallistic.z)) low = mid;
19806:           else high = mid;
19807:         }
19808:         ballisticPoint(origin, velocity, high, out);
19809:         return { time: high, x: out.x, y: out.y, z: out.z };
